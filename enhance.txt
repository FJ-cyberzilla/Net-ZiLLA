​You can add a simple helper module in Python using your standard requests library to query Cloudflare's security center endpoints:



import os
import requests

CF_API_TOKEN = os.getenv("CLOUDFLARE_API_TOKEN")
ACCOUNT_ID = os.getenv("CLOUDFLARE_ACCOUNT_ID")

def query_cloudflare_domain_intel(domain_name: str):
    url = f"https://api.cloudflare.com/client/v4/accounts/{ACCOUNT_ID}/intel/domain"
    headers = {
        "Authorization": f"Bearer {CF_API_TOKEN}",
        "Content-Type": "application/json"
    }
    params = {"domain": domain_name}
    
    response = requests.get(url, headers=headers, params=params)
    if response.status_code == 200:
        return response.json().get("result", {})
    else:
        return {"error": response.status_code, "message": response.text}
